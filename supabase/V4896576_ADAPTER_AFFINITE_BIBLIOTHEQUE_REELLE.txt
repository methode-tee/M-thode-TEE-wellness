-- MÉTHODE TEE — V4896576
-- Adapter mon repas : affinité culinaire réelle -> même bibliothèque que le Carnet.
-- Le moteur culinaire existant fournit des termes compatibles ; le serveur les
-- résout avec search_foods_v4 dans UN seul RPC. Aucun catalogue d'associations
-- codé en dur dans Adapter, aucune IA externe, aucune recherche Internet.

begin;

create or replace function public.mt_adapter_library_bridge_v3(
  p_meal_id uuid default null,
  p_input_text text default '',
  p_meal_type text default null,
  p_goal text default 'autre',
  p_culinary_terms text[] default '{}'::text[],
  p_culinary_hints text[] default '{}'::text[]
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $fn$
declare
  uid uuid:=auth.uid();
  v_segments jsonb:='[]'::jsonb;
  v_linked jsonb:='[]'::jsonb;
  v_pools jsonb:='{}'::jsonb;
  v_memory jsonb:='{}'::jsonb;
  v_excluded text[]:='{}'::text[];
  v_familiar uuid[]:='{}'::uuid[];
begin
  if uid is null then raise exception 'auth required'; end if;

  if to_regclass('public.mt_planner_preferences') is not null then
    select coalesce(excluded_terms,'{}'::text[])
      into v_excluded
    from public.mt_planner_preferences
    where user_id=uid;
    v_excluded:=coalesce(v_excluded,'{}'::text[]);
  end if;

  if to_regprocedure('public.mt_planner_personal_memory_v1(integer)') is not null then
    begin
      v_memory:=coalesce(public.mt_planner_personal_memory_v1(60),'{}'::jsonb);
      select coalesce(array_agg((x->>'food_dictionary_id')::uuid),'{}'::uuid[])
        into v_familiar
      from jsonb_array_elements(coalesce(v_memory->'familiar_dictionary_ids','[]'::jsonb)) x
      where nullif(x->>'food_dictionary_id','') is not null;
    exception when others then
      v_memory:='{}'::jsonb;
      v_familiar:='{}'::uuid[];
    end;
  end if;

  -- 1) Repas déjà enregistré : food_meal_items reste la source de vérité.
  if p_meal_id is not null then
    with x as (
      select i.*,
        d.display_name,d.canonical_name,d.country,d.culture,
        coalesce(d.categories,'{}'::text[]) categories,
        coalesce(d.typical_components,'[]'::jsonb) typical_components,
        coalesce(d.optional_components,'[]'::jsonb) optional_components,
        coalesce(d.adapter_profile,'{}'::jsonb) adapter_profile,
        c.name ciqual_name,
        case
          when to_regprocedure('public.mt_ciqual_role_v1(text,numeric,numeric,numeric,numeric,numeric)') is not null
          then public.mt_ciqual_role_v1(
            coalesce(c.name,i.food_name),
            coalesce(i.kcal_100g,c.kcal_100g),
            coalesce(i.protein_100g,c.protein_100g),
            coalesce(i.fat_100g,c.fat_100g),
            coalesce(i.carbs_100g,c.carbs_100g),
            coalesce(i.fiber_100g,c.fiber_100g)
          )
          else null
        end ciqual_role
      from public.food_meal_items i
      join public.food_meals m on m.id=i.meal_id and m.user_id=uid
      left join public.food_dictionary d on d.id=i.food_dictionary_id
      left join public.ciqual_foods c on c.code=coalesce(i.ciqual_code,d.ciqual_code)
      where i.meal_id=p_meal_id
    )
    select coalesce(jsonb_agg(jsonb_build_object(
      'food_name',food_name,
      'dictionary_id',food_dictionary_id,
      'ciqual_code',ciqual_code,
      'quantity_g',quantity_g,
      'kcal',kcal,'protein',protein,'fat',fat,'carbs',carbs,'fiber',fiber,'salt',salt,
      'kcal_100g',kcal_100g,'protein_100g',protein_100g,'fat_100g',fat_100g,
      'carbs_100g',carbs_100g,'fiber_100g',fiber_100g,'salt_100g',salt_100g,
      'micronutrients_100g',coalesce(micronutrients_100g,'{}'::jsonb),
      'nutrition_extra_100g',coalesce(nutrition_extra_100g,'{}'::jsonb),
      'categories',categories,
      'typical_components',typical_components,
      'optional_components',optional_components,
      'adapter_profile',adapter_profile,
      'roles',public.mt_adapt_roles_v1(
        coalesce(display_name,canonical_name,food_name,ciqual_name),
        categories,
        ciqual_role
      )
    ) order by sort_order),'[]'::jsonb)
    into v_linked
    from x;
  end if;

  -- 2) Même resolver que Ma journée alimentaire pour les composants saisis.
  with segs as (
    select row_number() over() ord,
      btrim(x) input,
      btrim(regexp_replace(
        btrim(x),
        '[0-9]+([.,][0-9]+)?[[:space:]]*(g|gr|grammes?|ml)',
        ' ',
        'gi'
      )) query_input
    from regexp_split_to_table(
      left(coalesce(p_input_text,''),1000),
      '\s*(?:\+|,|;|/|&)\s*|\s+et\s+|\s+avec\s+',
      'i'
    ) x
    where length(public.food_normalize(x))>=2
    limit 10
  ),
  c as (
    select s.ord,s.input,f.*,
      case
        when to_regprocedure('public.mt_ciqual_role_v1(text,numeric,numeric,numeric,numeric,numeric)') is not null
        then public.mt_ciqual_role_v1(
          coalesce(f.display_name,f.name),
          f.kcal_100g,f.protein_100g,f.fat_100g,f.carbs_100g,f.fiber_100g
        )
        else null
      end ciqual_role
    from segs s
    left join lateral public.search_foods_v4(
      case
        when length(public.food_normalize(s.query_input))>=2 then s.query_input
        else s.input
      end,
      4
    ) f on true
  ),
  grouped as (
    select s.ord,s.input,
      coalesce((
        select jsonb_agg(jsonb_build_object(
          'code',c.code,
          'name',c.name,
          'dictionary_id',c.dictionary_id,
          'display_name',c.display_name,
          'country',c.country,
          'categories',coalesce(c.categories,'{}'::text[]),
          'adapter_profile',coalesce(c.adapter_profile,'{}'::jsonb),
          'match_rank',c.match_rank,
          'kcal_100g',c.kcal_100g,
          'protein_100g',c.protein_100g,
          'fat_100g',c.fat_100g,
          'carbs_100g',c.carbs_100g,
          'fiber_100g',c.fiber_100g,
          'salt_100g',c.salt_100g,
          'micronutrients_100g',coalesce(c.micronutrients_100g,'{}'::jsonb),
          'nutrition_extra_100g',coalesce(c.nutrition_extra_100g,'{}'::jsonb),
          'roles',public.mt_adapt_roles_v1(
            coalesce(c.display_name,c.name),
            coalesce(c.categories,'{}'::text[]),
            c.ciqual_role
          ),
          'source',c.source
        ) order by c.match_rank,c.display_name,c.name)
        from c
        where c.ord=s.ord
          and (c.code is not null or c.dictionary_id is not null)
      ),'[]'::jsonb) candidates
    from segs s
  )
  select coalesce(
    jsonb_agg(jsonb_build_object('input',input,'candidates',candidates) order by ord),
    '[]'::jsonb
  )
  into v_segments
  from grouped;

  -- 3) Affinité culinaire : le moteur déjà chargé côté app fournit seulement
  --    des termes possibles (ex. champignon / brocoli / épinard).
  --    Ces termes sont alors résolus par search_foods_v4, donc par LA MÊME
  --    bibliothèque que le Carnet. Rien n'est choisi depuis une liste parallèle.
  with terms as (
    select ordinality::int ord,btrim(term) term
    from unnest(coalesce(p_culinary_terms,'{}'::text[]))
      with ordinality as t(term,ordinality)
    where length(public.food_normalize(term))>=2
    limit 8
  ),
  hits0 as (
    select
      t.ord,t.term,
      f.*,
      d.meal_contexts,
      d.priority,
      case when f.dictionary_id=any(v_familiar) then 1 else 0 end familiar,
      case
        when to_regprocedure('public.mt_ciqual_role_v1(text,numeric,numeric,numeric,numeric,numeric)') is not null
        then public.mt_ciqual_role_v1(
          coalesce(f.display_name,f.name),
          f.kcal_100g,f.protein_100g,f.fat_100g,f.carbs_100g,f.fiber_100g
        )
        else null
      end ciqual_role
    from terms t
    cross join lateral public.search_foods_v4(t.term,5) f
    left join public.food_dictionary d on d.id=f.dictionary_id
    where f.code is not null or f.dictionary_id is not null
  ),
  hits as (
    select h.*,
      public.mt_adapt_roles_v1(
        coalesce(h.display_name,h.name),
        coalesce(h.categories,'{}'::text[]),
        h.ciqual_role
      ) roles
    from hits0 h
    where not exists (
      select 1
      from unnest(v_excluded) e
      where length(public.food_normalize(e))>=2
        and public.food_normalize(coalesce(h.display_name,h.name))
            like '%'||public.food_normalize(e)||'%'
    )
      and public.food_normalize(coalesce(h.display_name,h.name))
          !~ '(^| )(bebe|infantile|premature|substitut de repas|poudre a reconstituer)( |$)'
  ),
  role_hits as (
    select h.*,r.role_key
    from hits h
    cross join lateral unnest(h.roles) as r(role_key)
    where r.role_key in ('protein','vegetable','starch','fruit')
  ),
  dedup as (
    select r.*,
      row_number() over(
        partition by r.role_key,
          coalesce(
            r.dictionary_id::text,
            nullif(r.code::text,''),
            public.food_normalize(coalesce(r.display_name,r.name))
          )
        order by
          r.ord,
          coalesce(r.match_rank,999),
          r.familiar desc,
          coalesce(r.priority,100),
          public.food_normalize(coalesce(r.display_name,r.name))
      ) duplicate_rank
    from role_hits r
  ),
  scored as (
    select d.*,
      (
        (d.ord-1)*7.0
        + least(35,coalesce(d.match_rank,35))
        + case when d.dictionary_id is null then 6 else 0 end
        + case when d.familiar=1 then -10 else 0 end
        + case
            when p_meal_type is not null
             and p_meal_type=any(coalesce(d.meal_contexts,'{}'::text[]))
            then -6 else 0
          end
        + coalesce(d.priority,100)/40.0
        + case
            when p_goal='prise_masse'
             and d.role_key='protein'
            then -least(8,coalesce(d.protein_100g,0)/4)
            else 0
          end
        + case
            when p_goal='energie'
             and d.role_key='starch'
            then -least(8,coalesce(d.carbs_100g,0)/10)
            else 0
          end
      ) score,
      greatest(
        1,
        100
        - least(70,coalesce(d.match_rank,70))
        - (d.ord-1)*8
      )::numeric culinary_affinity
    from dedup d
    where d.duplicate_rank=1
  ),
  ranked as (
    select s.*,
      row_number() over(
        partition by s.role_key
        order by
          s.score,
          s.ord,
          coalesce(s.match_rank,999),
          s.familiar desc,
          public.food_normalize(coalesce(s.display_name,s.name))
      ) rn
    from scored s
  ),
  picked as (
    select
      r.role_key,
      jsonb_build_object(
        'code',r.code,
        'name',r.name,
        'dictionary_id',r.dictionary_id,
        'display_name',r.display_name,
        'country',r.country,
        'categories',coalesce(r.categories,'{}'::text[]),
        'adapter_profile',coalesce(r.adapter_profile,'{}'::jsonb),
        'roles',to_jsonb(r.roles),
        'kcal_100g',r.kcal_100g,
        'protein_100g',r.protein_100g,
        'fat_100g',r.fat_100g,
        'carbs_100g',r.carbs_100g,
        'fiber_100g',r.fiber_100g,
        'salt_100g',r.salt_100g,
        'micronutrients_100g',coalesce(r.micronutrients_100g,'{}'::jsonb),
        'nutrition_extra_100g',coalesce(r.nutrition_extra_100g,'{}'::jsonb),
        'familiar',r.familiar=1,
        'score',round(r.score::numeric,2),
        'culinary_affinity',round(r.culinary_affinity,2),
        'culinary_term',r.term,
        'source',r.source
      ) obj,
      r.rn
    from ranked r
    where r.rn<=8
  )
  select coalesce(jsonb_object_agg(role_key,arr),'{}'::jsonb)
    into v_pools
  from (
    select role_key,jsonb_agg(obj order by rn) arr
    from picked
    group by role_key
  ) z;

  return jsonb_build_object(
    'version','V4896576_LIBRARY_AFFINITY',
    'resolver','search_foods_v4 + food_meal_items snapshots',
    'segments',coalesce(v_segments,'[]'::jsonb),
    'linked_items',coalesce(v_linked,'[]'::jsonb),
    'candidate_pools',coalesce(v_pools,'{}'::jsonb),
    'culinary_hints',to_jsonb(coalesce(p_culinary_hints,'{}'::text[])),
    'culinary_terms',to_jsonb(coalesce(p_culinary_terms,'{}'::text[])),
    'excluded_terms',to_jsonb(coalesce(v_excluded,'{}'::text[])),
    'rules',jsonb_build_object(
      'same_resolver_as_food_day',true,
      'saved_meal_snapshot_is_authoritative',true,
      'recognition_is_not_quantity',true,
      'culinary_engine_before_library_choice',true,
      'culinary_terms_resolved_by_search_foods_v4',true,
      'no_hardcoded_food_pair_catalog_in_adapter',true,
      'fail_closed_when_no_compatible_candidate',true,
      'profile_memory_ranks_only_compatible_candidates',true,
      'no_external_ai',true,
      'no_internet_lookup',true
    )
  );
end;
$fn$;

revoke all on function public.mt_adapter_library_bridge_v3(
  uuid,text,text,text,text[],text[]
) from public,anon;

grant execute on function public.mt_adapter_library_bridge_v3(
  uuid,text,text,text,text[],text[]
) to authenticated;

commit;

  'culinary_engine_before_library_choice',true,

  'external_ai',false,

  'payments_changed',false,
  'unlock_changed',false
) as v4896576_result;
